# wasi:http, the same requests from the guest in Wasmtime and from the host of wasmext, to the server of server.py (PORT is its port)
# req METHOD AUTHORITY PATH [body=TEXT] [h:NAME=VALUE] [tr:NAME=VALUE] [first=MS]
req GET 127.0.0.1:PORT /hello
req GET 127.0.0.1:PORT /hello h:x-test=1
req POST 127.0.0.1:PORT /echo?a=b body=ping h:content-type=text/x
req POST 127.0.0.1:PORT /echo body=data tr:x-end=yes
req PUT 127.0.0.1:PORT /echo body=put
req DELETE 127.0.0.1:PORT /echo
req POST 127.0.0.1:PORT /echo
req PROPFIND 127.0.0.1:PORT /echo body=x
req HEAD 127.0.0.1:PORT /hello
req GET 127.0.0.1:PORT /chunked
req GET 127.0.0.1:PORT /status/404
req GET 127.0.0.1:PORT /status/500
req GET 127.0.0.1:PORT /info
req GET 127.0.0.1:PORT /closeend
req GET 127.0.0.1:PORT /truncated  #~ Wasmtime streams the body: the answer comes at once and the stream fails at the end; here the answer is read whole, so the request fails (HTTP-response-incomplete)
req GET 127.0.0.1:PORT /slow first=100
req GET 127.0.0.1:PORT /badstatus
req GET 127.0.0.1:PORT /te  #~ an unknown transfer coding: Wasmtime (hyper) reads to the end of the stream; here HTTP-response-transfer-coding, as the WIT has it
req GET 127.0.0.1:PORT /badchunk  #~ as /truncated: Wasmtime answers and the body stream fails; here the request fails (HTTP-protocol-error)
req GET 127.0.0.1:PORT /bighead  #~ a head of 70 000 bytes: Wasmtime takes it, here the limit is 64 KiB (HTTP-response-header-section-size)
req GET 127.0.0.1:PORT /nothing
req GET 127.0.0.1:1 /
req GET nonexistent.invalid /
